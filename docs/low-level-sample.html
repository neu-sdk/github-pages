Low-Level API サンプルアプリ

操作ガイド

  1. 装置を起動させて、PCとBluetooth かUSB 接続させて下さい。
  2. 装置を装着してください。センサーユニットが眉の上に来るように装着し、髪の毛が挟まらないように注意して下さい。具体的な装着方法については、取扱説明書(リンク)をご覧ください。
  3. PowerShell を開いていただき、LowLevelSample.exeを実行してください。処理がすべて完了すると「Press any key to quit」と表示され、キー入力を受け取るまでアプリは終了しません。これにより、デフォルトのターミナルから実行した場合でもウィンドウが即座に閉じてしまうことを防げます。
  4. 実行開始すると、API 初期化、装置検出・選択・接続、（任意）ゲイン調整、（任意）キャリブレーション、計測開始、計測停止、装置切断の順に処理が進行します。

また、装置の選択時とゲイン・キャリブレーション開始時、計測開始時、計測終了時の進行をコマンドインターフェースで進める仕様になっております。

コード全体

このセクションでは、Low-Level API を利用して計測処理を一通り実行するサンプルアプリのコードを掲載しています。

using ExBrainSdk;
using ExBrainSdk.Datas;
using ExBrainSdk.Enumerations;

internal class Program
{
    private static void Main()
    {
        var api = new ExBrainApi((_, s) => Console.WriteLine("C# Status " + s), (id, message, type) =>
        {
            Console.WriteLine($"C# Log: [{id}] [{type}] {message}");
        });

        Sample(api).Wait();
        api.ExitApi();

        Console.WriteLine("Low-Level Sample finished");
        Console.WriteLine("Press any key to quit");
        _ = Console.ReadKey();
    }

    public static async Task Sample(ExBrainApi api)
    {
        EnmSdkResult sdkResult = api.InitApi(
            needOutputTraceLog: true,
            needOutputCommLog: false, // Not implemented
            logDir: null,
            settingsDir: null, // Not implemented
            tag: "" // Not implemented
        );
        if (sdkResult != EnmSdkResult.eSuccess)
        {
            Console.WriteLine("Failed to initialize API.");
            return;
        }

        await api.ResetBluetoothModuleAsync();

        var searchParam = new ExBrainSearchParameter()
        {
            Keywords = ["HOT", "XB"],
            ShouldUSB = true,
            ShouldPairedBluetooth = true,
            ShouldNonPairedBluetooth = true,
            TimeoutSecond = 3
        };
        api.StartSearchDevice(searchParam);
        await Task.Delay(3000);
        api.StopSearchDevice();
        if (api.DiscoveredDevice.DeviceCount == 0)
        {
            Console.WriteLine("No devices found.");
            return;
        }
        else
        {
            IReadOnlyList<DeviceInformationEx> devicesInfo = api.DiscoveredDevice.Devices;
            for (int i = 0; i < devicesInfo.Count; i++)
            {
                Console.WriteLine($"Find Device #{i}: Type: {devicesInfo[i].ConnectType}, DeviceId: {devicesInfo[i].DeviceId}");
            }
            Console.WriteLine("Please select a device index to connect:");
            string? index = await Console.In.ReadLineAsync();
            if (!int.TryParse(index, out int selectedIndex) || selectedIndex < 0 || selectedIndex >= devicesInfo.Count)
            {
                Console.WriteLine("Invalid index.");
                return;
            }
            else
            {
                DeviceInformationEx selectedDeviceInfo = devicesInfo[selectedIndex];
                sdkResult = await api.ConnectDeviceAsync(selectedDeviceInfo.DeviceId, selectedDeviceInfo.ConnectType);
                if (sdkResult != EnmSdkResult.eSuccess)
                {
                    Console.WriteLine("Failed to connect to device.");
                    return;
                }
            }
        }

        do
        {
            Console.WriteLine("Perform Gain Adjustment? (y/n):");
            string? input = await Console.In.ReadLineAsync();
            if (input?.ToLower() == "y")
            {
                sdkResult = await api.PerformGainAdjustmentAsync(api.HandleId);
                if (sdkResult != EnmSdkResult.eSuccess)
                {
                    Console.WriteLine("Gain adjustment failed.");
                    await api.DisconnectDeviceAsync();
                    return;
                }
                Console.WriteLine("Gain adjustment completed.");
                break;
            }
            else if (input?.ToLower() == "n")
            {
                break;
            }
        }
        while (true);

        do
        {
            Console.WriteLine("Perform Calibration? (y/n):");
            string? input = await Console.In.ReadLineAsync();
            if (input?.ToLower() == "y")
            {
                sdkResult = await api.PerformCalibrationAsync(api.HandleId);
                if (sdkResult != EnmSdkResult.eSuccess)
                {
                    Console.WriteLine("Calibration failed.");
                    await api.DisconnectDeviceAsync();
                    return;
                }
                Console.WriteLine("Calibration completed.");
                break;
            }
            else if (input?.ToLower() == "n")
            {
                break;
            }
        }
        while (true);

        EnmSdkMeasurementCycle measurementCycle;
        do
        {
            Console.WriteLine("Select measurement cycle:");
            Console.WriteLine("  0: 10Hz");
            Console.WriteLine("  1: 50Hz");
            Console.WriteLine("  2: 100Hz");
            Console.WriteLine("  3: 500Hz");
            string? input = await Console.In.ReadLineAsync();
            if (input == "0")
            {
                measurementCycle = EnmSdkMeasurementCycle.e10Hz;
                break;
            }
            if (input == "1")
            {
                measurementCycle = EnmSdkMeasurementCycle.e50Hz;
                break;
            }
            if (input == "2")
            {
                measurementCycle = EnmSdkMeasurementCycle.e100Hz;
                break;
            }
            if (input == "3")
            {
                measurementCycle = EnmSdkMeasurementCycle.e500Hz;
                break;
            }

            Console.WriteLine("Invalid selection. Please enter 0, 1, 2, or 3.");
        }
        while (true);

        EnmSdkMeasurementMode measurementMode = EnmSdkMeasurementMode.eNormal;
        if (measurementCycle != EnmSdkMeasurementCycle.e10Hz)
        {
            do
            {
                Console.WriteLine("Select measurement mode:");
                Console.WriteLine("  0: Low Precision / High Reliability");
                Console.WriteLine("  1: High Precision / Low Reliability");
                string? input = await Console.In.ReadLineAsync();
                if (input == "0")
                {
                    measurementMode = EnmSdkMeasurementMode.eLowPrecisionHighReliability;
                    break;
                }
                if (input == "1")
                {
                    measurementMode = EnmSdkMeasurementMode.eHighPrecisionLowReliability;
                    break;
                }

                Console.WriteLine("Invalid selection. Please enter 0 or 1.");
            }
            while (true);
        }

        List<ExBrainMeasureData> frames = [];
        sdkResult = await api.StartMeasureAsync(api.HandleId, "./", (_, data) =>
        {
            frames.Add(data);
        }, measurementCycle: measurementCycle, measurementMode: measurementMode);
        if (sdkResult != EnmSdkResult.eSuccess)
        {
            Console.WriteLine("Failed to start measurement.");
            await api.DisconnectDeviceAsync();
            return;
        }

        do
        {
            Console.WriteLine("Stop Measurement? (y/n):");
            string? input = await Console.In.ReadLineAsync();
            if (input?.ToLower() == "y")
            {
                break;
            }
        }
        while (true);

        sdkResult = await api.StopMeasureAsync(api.HandleId);
        if (sdkResult != EnmSdkResult.eSuccess)
        {
            Console.WriteLine("Failed to stop measurement.");
        }
        await api.DisconnectDeviceAsync();
    }
}

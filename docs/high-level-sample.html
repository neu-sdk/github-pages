High-Level API サンプルアプリ

操作ガイド

  1. 装置を起動させて、PC と Bluetooth か USB 接続させて下さい。接続ができていないと計測開始の工程へ進められない仕様になっているため、安定しやすい USB 接続を推奨します。
  2. 装置を装着してください。センサーユニットが眉の上に来るように装着し、髪の毛が挟まらないように注意して下さい。具体的な装着方法については、取扱説明書（リンク）をご覧ください。
  3. PowerShell を開いていただき、HighLevelSample.exe を実行してください。処理がすべて完了すると「Press any key to quit」と表示され、キー入力を受け取るまでアプリは終了しません。これにより、デフォルトのターミナルから実行した場合でもウィンドウが即座に閉じてしまうことを防げます。
  4. 実行すると、API 初期化、デバイス接続、ゲイン調整、キャリブレーション、実験開始、複数ブロック計測、統計比較、実験終了、デバイス切断までが順に実行されます。

コード全体

このセクションでは、High-Level API を利用して計測処理を一通り実行するサンプルアプリのコードを掲載しています。 コードは、デバイス接続からゲイン調整・キャリブレーション、実験の開始、ブロック実行、統計比較、そして計測終了までの流れを段階的に示しており、実際のアプリケーション開発の基盤となる構造を確認できます。

ここで示されるコードは、後続セクション「各パートのコード解説」で詳しく分解しながら解説します。 まずは全体像を把握するために、本セクションを参照してください。

Program.cs

using ExBrainSdk;
using ExBrainSdk.Enumerations;

var api = new ExBrainApi((_, s) => Console.WriteLine("C# Status " + s), (id, message, type) =>
{
    Console.WriteLine($"C# Log: [{id}] [{type}] {message}");
});

_ = api.InitApi(true, false, null, null);
if (await api.SearchAndConnectDeviceAsync(EnmSdkConnectType.eNotSelected, 3) == EnmSdkResult.eSuccess)
{
    await HighLevelSample.Run(api);
    Console.WriteLine("High-Level Sample finished");
}
else
{
    Console.WriteLine("Failed to connect to device.");
}
api.ExitApi();

Console.WriteLine("Press any key to quit");
Console.ReadKey();

HighLevelSample.cs

using ExBrainSdk;
using ExBrainSdk.HighLevel;

public static class HighLevelSample
{
    public static async Task Run(ExBrainApi api, Action<string>? log = null, bool performGainAdjustment = true, bool performCalibration = true, Experiment.NotifyInstantFrameCallback? instantFrameCallback = null, string? outputFolder = "")
    {
        log ??= Console.WriteLine;

        if (performGainAdjustment)
        {
            _ = await api.PerformGainAdjustmentAsync(api.HandleId);
            log("Gain adjustment completed.");
        }
        if (performCalibration)
        {
            _ = await api.PerformCalibrationAsync(api.HandleId);
            log("Calibration completed.");
        }

        var experiment = new Experiment(api, 10, instantFrameCallback);
        _ = experiment.SetLowpassFilter(2, 0.2, 10);
        _ = await experiment.StartExperimentAsync(outputFolder);
        log("Experiment started. Idling for 5 seconds...");
        await Task.Delay(TimeSpan.FromSeconds(5));
        log("Idling finished");

        int preFittingSeconds = 5;
        int taskSeconds = 5;
        int calmDownSeconds = 5;
        int postFittingSeconds = 5;
        var preTaskMetaData = new MetaData(api, "TestTask", preFittingSeconds, taskSeconds, calmDownSeconds, postFittingSeconds);
        var postTaskMetaData = new MetaData(api, "TestTask", preFittingSeconds, taskSeconds, calmDownSeconds, postFittingSeconds);
        int totalSeconds = preFittingSeconds + taskSeconds + calmDownSeconds + postFittingSeconds;

        string outputDir = outputFolder ?? "";
        log($"Starting block1 ({totalSeconds} seconds)");
        Experiment.NotifyStageStartedCallback callback = stage =>
        {
            switch (stage)
            {
                case BlockStage.PreFitting:
                    break;
                case BlockStage.Task:
                    log("TASK");
                    break;
                case BlockStage.CalmDown:
                    log("REST");
                    break;
                case BlockStage.PostFitting:
                    break;
                default:
                    break;
            }
        };
        log("REST");
        Block block1 = await experiment.RunBlockAsync(preTaskMetaData, "TestBlock", "Participant1", callback);
        log("Block1 finished");
        await File.WriteAllTextAsync(Path.Combine(outputDir, "block1.json"), block1.ToJson());
        Block block1_reconstructed = Block.FromJson(api, block1.ToJson())!;
        await File.WriteAllTextAsync(Path.Combine(outputDir, "block1.csv"), block1_reconstructed.ToCsv());

        log($"Starting block2 ({totalSeconds} seconds)");
        Block block2 = await experiment.RunBlockAsync(preTaskMetaData, "TestBlock", "Participant1", callback);
        log("Block2 finished");
        await File.WriteAllTextAsync(Path.Combine(outputDir, "block2.json"), block2.ToJson());

        log($"Starting block3 ({totalSeconds} seconds)");
        Block block3 = await experiment.RunBlockAsync(preTaskMetaData, "TestBlock", "Participant1", callback);
        log("Block3 finished");
        await File.WriteAllTextAsync(Path.Combine(outputDir, "block3.json"), block3.ToJson());

        // Add your own intervention codes.

        log($"Starting block4 ({totalSeconds} seconds)");
        Block block4 = await experiment.RunBlockAsync(postTaskMetaData, "TestBlock", "Participant1", callback);
        log("Block4 finished");
        await File.WriteAllTextAsync(Path.Combine(outputDir, "block4.json"), block4.ToJson());

        log($"Starting block5 ({totalSeconds} seconds)");
        Block block5 = await experiment.RunBlockAsync(postTaskMetaData, "TestBlock", "Participant1", callback);
        log("Block5 finished");
        await File.WriteAllTextAsync(Path.Combine(outputDir, "block5.json"), block5.ToJson());

        log($"Starting block6 ({totalSeconds} seconds)");
        Block block6 = await experiment.RunBlockAsync(postTaskMetaData, "TestBlock", "Participant1", callback);
        log("Block6 finished");
        await File.WriteAllTextAsync(Path.Combine(outputDir, "block6.json"), block6.ToJson());

        _ = await experiment.FinishExperimentAsync();
        await api.DisconnectDeviceAsync();

        var comparison = TaskComparison.CompareWithinParticipantPrePost(api, [block1, block2, block3], [block4, block5, block6]);
        log($"The statistical test being used is: {comparison.dTHbL_left_testKind}");
        log("Comparison for dTHbL(leftSD3-1cm)Processed");
        log($"  P-value = {comparison.dTHbL_left_pValue}, Significant = {comparison.dTHbL_left_pValue < 0.05}");
    }
}

各パートのコード解説

続いて、各パート毎のコード解説を行っていきます。 このサンプルでは、トップレベルの Program.cs で初期化と接続を行い、その後 HighLevelSample.Run(...) で計測処理全体を実行します。

1. Program.cs の内容

using ExBrainSdk;
using ExBrainSdk.Enumerations;

var api = new ExBrainApi((_, s) => Console.WriteLine("C# Status " + s), (id, message, type) =>
{
    Console.WriteLine($"C# Log: [{id}] [{type}] {message}");
});

_ = api.InitApi(true, false, null, null);
if (await api.SearchAndConnectDeviceAsync(EnmSdkConnectType.eNotSelected, 3) == EnmSdkResult.eSuccess)
{
    await HighLevelSample.Run(api);
    Console.WriteLine("High-Level Sample finished");
}
else
{
    Console.WriteLine("Failed to connect to device.");
}
api.ExitApi();

Console.WriteLine("Press any key to quit");
Console.ReadKey();

1.1 ExBrainApi の生成

最初に ExBrainApi のコンストラクタ を呼び出し、API インスタンスを生成しています。 コンストラクタにはステータス変化通知用コールバックとログ出力用コールバックを渡しており、SDK 内部の状態変化やメッセージをコンソールへ表示できるようにしています。

var api = new ExBrainApi((_, s) => Console.WriteLine("C# Status " + s), (id, message, type) =>
{
    Console.WriteLine($"C# Log: [{id}] [{type}] {message}");
});

これにより、接続や計測の進行状態をサンプル実行中に確認できます。

1.2 API の初期化

次に InitApi メソッド を呼び出し、SDK を初期化します。 このサンプルでは戻り値を利用しないため、discard である _ に代入しています。

_ = api.InitApi(true, false, null, null);

引数では、トレースログ出力を有効にし、通信ログ出力や追加のディレクトリ設定は省略しています。 実運用アプリケーションでは、戻り値を確認して失敗時のエラー処理を実装することを推奨します。

1.3 デバイスの検索と接続

初期化後は SearchAndConnectDeviceAsync メソッド を呼び出して、デバイスの検索と接続を行います。

if (await api.SearchAndConnectDeviceAsync(EnmSdkConnectType.eNotSelected, 3) == EnmSdkResult.eSuccess)
{
    // 計測処理（1.4 で解説）
}
else
{
    Console.WriteLine("Failed to connect to device.");
}

ここでは EnmSdkConnectType.eNotSelected を指定しているため、接続方式を固定せず、利用可能な接続手段から接続を試みる構成になっています。 第 2 引数 3 は検索・接続処理に関するタイムアウト秒数です。 接続に成功した場合のみ計測処理（1.4）を実行し、失敗した場合はメッセージを表示します。 いずれの場合も後続の終了処理（1.5）に進み、キー入力を待ってからアプリが終了するため、接続に失敗してもエラーメッセージを確認できます。

1.4 実験本体の実行

デバイス接続が成功した場合（1.3 の if ブロック内）、実験本体である HighLevelSample.Run(api) を呼び出します。

await HighLevelSample.Run(api);
Console.WriteLine("High-Level Sample finished");

ここでは HighLevelSample.Run(api) を呼び出すことで、計測に関する一連の処理を実行し、完了後に終了メッセージを表示しています。

1.5 API の終了処理

最後に ExitApi メソッド を呼び出して API をクリーンに終了します。

api.ExitApi();

Console.WriteLine("Press any key to quit");
Console.ReadKey();

api.ExitApi() は接続の成否にかかわらず（1.3 の if／else のどちらを通った場合でも）呼び出され、初期化時に確保された内部リソースを解放します。 その後 Console.ReadKey() でキー入力を待ってからアプリが終了します。これを最後に置くことで、ターミナルから実行した際にウィンドウが即座に閉じてしまうのを防ぎ、接続失敗時のエラーメッセージも確認できるようにしています。

2.1 ログ出力先の決定

関数冒頭では、呼び出し元からログ出力先が渡されていない場合に Console.WriteLine を使用するよう設定しています。

log ??= Console.WriteLine;

これにより、このサンプルはコンソールアプリとしてそのまま使える一方で、GUI アプリケーションやテストコードから別のログ出力先を差し込むこともできます。

2.2 ゲイン調整の実行

performGainAdjustment が true の場合、PerformGainAdjustmentAsync メソッド を呼び出してゲイン調整を行います。

if (performGainAdjustment)
{
    _ = await api.PerformGainAdjustmentAsync(api.HandleId);
    log("Gain adjustment completed.");
}

api.HandleId は接続済みデバイスを識別するためのハンドルです。 ゲイン調整は、計測値が適切なレンジに収まるよう補正するための前処理として利用されます。 このサンプルでは戻り値を検証していませんが、実際のアプリケーションでは結果コードに基づく分岐処理を追加するのが一般的です。

2.3 キャリブレーションの実行

次に performCalibration が true の場合、PerformCalibrationAsync メソッド を呼び出してキャリブレーションを行います。

if (performCalibration)
{
    _ = await api.PerformCalibrationAsync(api.HandleId);
    log("Calibration completed.");
}

キャリブレーションは装着状態や個人差を反映した補正処理です。 ゲイン調整と同様に、このサンプルでは正常系の流れを分かりやすく示すため簡潔な記述にしています。

2.4 Experiment インスタンスの生成とフィルタ設定

計測本体は Experiment クラス によって管理されます。 ここでは Experiment(api, 10, instantFrameCallback) を生成し、その後ローパスフィルタを設定しています。

var experiment = new Experiment(api, 10, instantFrameCallback);
_ = experiment.SetLowpassFilter(2, 0.2, 10);

第 2 引数 10 は SMD 計算用のターゲットウィンドウサイズです。 instantFrameCallback を指定すれば、計測中のリアルタイムフレーム通知を受け取ることもできます。 また、SetLowpassFilter(2, 0.2, 10) では Butterworth ローパスフィルタを設定しています。 パラメータは順に、フィルタ次数、カットオフ周波数、サンプリングレートを表します。

2.5 実験の開始

フィルタ設定後、StartExperimentAsync メソッド を呼び出して計測を開始します。

_ = await experiment.StartExperimentAsync(outputFolder);
log("Experiment started. Idling for 5 seconds...");
await Task.Delay(TimeSpan.FromSeconds(5));
log("Idling finished");

outputFolder を指定すると、実験開始時に CSV 出力先として利用されます。 開始直後には 5 秒の待機時間を設けており、ここでは計測開始後の安定化期間や、ユーザーによる導入処理の余地を表現しています。

2.6 ブロック時間設定と MetaData の作成

次に、各ブロックで使用する時間設定を定義し、MetaData クラス を生成します。

int preFittingSeconds = 5;
int taskSeconds = 5;
int calmDownSeconds = 5;
int postFittingSeconds = 5;
var preTaskMetaData = new MetaData(api, "TestTask", preFittingSeconds, taskSeconds, calmDownSeconds, postFittingSeconds);
var postTaskMetaData = new MetaData(api, "TestTask", preFittingSeconds, taskSeconds, calmDownSeconds, postFittingSeconds);
int totalSeconds = preFittingSeconds + taskSeconds + calmDownSeconds + postFittingSeconds;

各値は以下の意味を持ちます。

  • preFittingSeconds: タスク前の事前装着調整フェーズ
  • taskSeconds: 実タスクフェーズ
  • calmDownSeconds: タスク後のクールダウンフェーズ
  • postFittingSeconds: 最後の事後装着調整フェーズ

このサンプルでは前半と後半で同一設定を使用していますが、preTaskMetaData と postTaskMetaData を分けることで、将来的に条件の異なるブロック構成へ拡張しやすくしています。

2.7 前半ブロックの実行と出力

最初の 3 ブロックは前半条件として実行されます。 各ブロックでは RunBlockAsync メソッド を呼び出し、完了後に JSON ファイルを保存します。 1 ブロック目のみ CSV も併せて出力しています。

string outputDir = outputFolder ?? "";
log($"Starting block1 ({totalSeconds} seconds)");
Experiment.NotifyStageStartedCallback callback = stage =>
{
    switch (stage)
    {
        case BlockStage.PreFitting:
            break;
        case BlockStage.Task:
            log("TASK");
            break;
        case BlockStage.CalmDown:
            log("REST");
            break;
        case BlockStage.PostFitting:
            break;
        default:
            break;
    }
};
log("REST");
Block block1 = await experiment.RunBlockAsync(preTaskMetaData, "TestBlock", "Participant1", callback);
log("Block1 finished");
await File.WriteAllTextAsync(Path.Combine(outputDir, "block1.json"), block1.ToJson());
Block block1_reconstructed = Block.FromJson(api, block1.ToJson())!;
await File.WriteAllTextAsync(Path.Combine(outputDir, "block1.csv"), block1_reconstructed.ToCsv());

log($"Starting block2 ({totalSeconds} seconds)");
Block block2 = await experiment.RunBlockAsync(preTaskMetaData, "TestBlock", "Participant1", callback);
log("Block2 finished");
await File.WriteAllTextAsync(Path.Combine(outputDir, "block2.json"), block2.ToJson());

log($"Starting block3 ({totalSeconds} seconds)");
Block block3 = await experiment.RunBlockAsync(preTaskMetaData, "TestBlock", "Participant1", callback);
log("Block3 finished");
await File.WriteAllTextAsync(Path.Combine(outputDir, "block3.json"), block3.ToJson());

RunBlockAsync(...) は、事前装着調整、タスク、クールダウン、事後装着調整を含む 1 ブロック分の処理を完了まで実行し、その結果を Block クラス として返します。 ToJson() と ToCsv() を使うことで、ブロック単位の結果をそのまま保存できます。

block1 では CSV を出力する前に Block.FromJson(api, block1.ToJson()) で JSON 経由のラウンドトリップを行い、シリアライズ／デシリアライズが正しく機能することを示しています。 通常のアプリケーションでは block1.ToCsv() を直接呼び出すだけで十分です。

第 4 引数の NotifyStageStartedCallback は BlockStage の各フェーズが開始するたびに PreFitting / Task / CalmDown / PostFitting の順で通知されます。 このサンプルでは callback という変数に一度だけ定義し、全 6 ブロックで使い回しています。switch 文で受け取ったステージごとに分岐し、Task 開始時（タスク開始時）に TASK、CalmDown 開始時（タスク終了時）に REST をログ出力することで、参加者に対するタスク／休憩の合図を表現しています。block1 の直前に log("REST") を出力しているのは、最初のブロック開始までの初期待機状態を REST として扱うためです。 コールバック自体は省略可能ですが、ここでは全ブロックで同一の挙動が必要なため明示的に渡しています。

2.8 介入処理の挿入ポイント

前半と後半のブロック群の間には、コメントで介入処理の挿入ポイントが示されています。

// Add your own intervention codes.

ここでは、前半ブロック群と後半ブロック群の間に、刺激提示、条件変更、外部タスク開始などの独自処理を差し込むことを想定しています。

2.9 後半ブロックの実行

後半の 3 ブロックも同様に RunBlockAsync(...) で実行し、結果を JSON として保存します。

log($"Starting block4 ({totalSeconds} seconds)");
Block block4 = await experiment.RunBlockAsync(postTaskMetaData, "TestBlock", "Participant1", callback);
log("Block4 finished");
await File.WriteAllTextAsync(Path.Combine(outputDir, "block4.json"), block4.ToJson());

log($"Starting block5 ({totalSeconds} seconds)");
Block block5 = await experiment.RunBlockAsync(postTaskMetaData, "TestBlock", "Participant1", callback);
log("Block5 finished");
await File.WriteAllTextAsync(Path.Combine(outputDir, "block5.json"), block5.ToJson());

log($"Starting block6 ({totalSeconds} seconds)");
Block block6 = await experiment.RunBlockAsync(postTaskMetaData, "TestBlock", "Participant1", callback);
log("Block6 finished");
await File.WriteAllTextAsync(Path.Combine(outputDir, "block6.json"), block6.ToJson());

後半ブロックも前半と同じ callback を使い回すことで、TASK / REST の合図表示を全ブロックで統一しています。 前半ブロック群と後半ブロック群を分けて保存することで、解析時に条件ごとのまとまりを維持しやすくなります。

2.10 実験の終了とデバイス切断

すべてのブロック実行後、FinishExperimentAsync メソッド で実験を終了し、その後 DisconnectDeviceAsync メソッド でデバイスを切断します。

_ = await experiment.FinishExperimentAsync();
await api.DisconnectDeviceAsync();

計測を終了する際には FinishExperimentAsync() を呼び出し、その後にデバイス切断処理を行います。

2.11 介入前後の統計比較

最後に、前半 3 ブロックと後半 3 ブロックを TaskComparison.CompareWithinParticipantPrePost メソッド で比較します。

var comparison = TaskComparison.CompareWithinParticipantPrePost(api, [block1, block2, block3], [block4, block5, block6]);
log($"The statistical test being used is: {comparison.dTHbL_left_testKind}");
log("Comparison for dTHbL(leftSD3-1cm)Processed");
log($"  P-value = {comparison.dTHbL_left_pValue}, Significant = {comparison.dTHbL_left_pValue < 0.05}");

この比較は、同一参加者の介入前後ブロック群を対象にした統計評価です。 ログには使用された検定種別と dTHbL(leftSD3-1cm)Processed の p 値、有意差判定結果を出力しています。 必要に応じて右側データや差分値の出力を追加することも可能です。

以上で、High-Level API サンプルアプリのソースコード解説を終わります。
